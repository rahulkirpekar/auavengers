--------------------------------------

4. NullPointerException:-
--------------------------------------

	SOP(name.length());// NullPointerException

	Student s = null;
	SOP(s.toString());

5.  NumberFormateException
------------------------------
 
	String value = "a1b2c3";

	int no1 = Integer.parseInt(value);// NumberFormateException

	SOP(no1+100);// 223

6.  ClassCastException
------------------------------

		A(test1())
		|
	-----------------
	|		|
	B		C
   test1()--B	     test1()--C

   testB();	     testC();


	// Upcasting
	A obj = new C();
	obj.test1();// B


	// obj.testB();// C.E


	// Downcasting
	B objB = (B)obj;// ClassCastException
	objB.testB();

	





	obj = new C();
	obj.test1();// C
